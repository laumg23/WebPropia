    <script>
        // 1. Inicializamos el mapa con una vista global por defecto
        const map = L.map('map').setView([20, 0], 2);

        // 2. Cargamos el mapa base limpio de Esri (Fondo beige/claro)
        L.tileLayer('https://arcgisonline.com{z}/{y}/{x}', {
            attribution: 'Tiles &copy; Esri &mdash; Source: US National Park Service',
            maxZoom: 8
        }).addTo(map);

        // 3. Superponemos las fronteras políticas y nombres de países de Esri
        L.tileLayer('https://arcgisonline.com{z}/{y}/{x}').addTo(map);

        // 4. Configuración del archivo de Strava
        // REQUISITO: Tu archivo descargado de Strava debe llamarse "mi_ruta.gpx" y estar en la misma carpeta
        const gpxUrl = 'mi_ruta.gpx'; 

        // 5. Procesamos el archivo GPX y lo dibujamos en el mapa
        const rutaStrava = new L.GPX(gpxUrl, {
            async: true,
            polyline_options: {
                color: '#ff3b30',     // Color rojo vivo idéntico al de la imagen
                weight: 4,            // Grosor de la línea
                opacity: 0.85         // Nitidez de la línea
            },
            marker_options: {
                startIconUrl: null,   // Ocultamos el pin verde por defecto de inicio de ruta
                endIconUrl: null,     // Ocultamos el pin rojo por defecto de fin de ruta
                shadowUrl: null
            }
        }).on('loaded', function(e) {
            // Cuando la ruta se lee con éxito, el mapa hace zoom automático sobre ella
            map.fitBounds(e.target.getBounds());
            
            // Extraemos los kilómetros reales grabados en tu Strava
            const distanciaMetros = e.target.get_distance();
            const distanciaKm = (distanciaMetros / 1000).toFixed(1);
            
            // Actualizamos la tarjeta de estadísticas de tu web automáticamente
            document.getElementById('contador-distancia').innerText = distanciaKm + " km";
            document.getElementById('estado-ruta').innerText = "Ruta sincronizada";

            // Colocamos el punto rojo parpadeante en tu posición actual (la última coordenada de la ruta)
            const coordenadas = e.target.get_layers()[0].getLatLngs();
            const ultimaCoordenada = coordenadas[coordenadas.length - 1];

            L.circleMarker(ultimaCoordenada, {
                radius: 7,
                fillColor: '#ff3b30',
                color: '#fff',
                weight: 2,
                opacity: 1,
                fillOpacity: 1
            }).addTo(map).bindPopup("<b>¡Mi última posición!</b>").openPopup();

        }).on('error', function(e) {
            // Si el archivo no se encuentra o está mal puesto, avisa en la tarjeta
            document.getElementById('estado-ruta').innerText = "Falta el archivo mi_ruta.gpx";
            document.getElementById('estado-ruta').style.color = "orange";
            console.error("No se pudo cargar el archivo GPX. Asegúrate de renombrarlo a mi_ruta.gpx");
        }).addTo(map);
    </script>
